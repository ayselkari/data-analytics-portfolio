-- Suggested Power BI measures

Total Bookings = COUNTROWS(Bookings)

Total Revenue = SUM(Bookings[Revenue_EUR])

Total Profit = SUM(Bookings[Profit_EUR])

Average Booking Value = DIVIDE([Total Revenue], [Total Bookings])

Cancellation Rate =
DIVIDE(
    CALCULATE([Total Bookings], Bookings[Status] = "Cancelled"),
    [Total Bookings]
)

Completed Revenue =
CALCULATE([Total Revenue], Bookings[Status] = "Completed")
